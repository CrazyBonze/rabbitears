# --- Playback: mpv + VAAPI hardware decode + DRM output (SCOPE §3, §4.3) ---
mpv

libva-intel-driver    # Cherry Trail is Gen8 -> the i965 VAAPI driver
libva-utils           # provides `vainfo` (M1 verifies VAAPI)

# --- App runtime: Python + evdev remote (SCOPE §3, §4.4) ---
python3
py3-evdev

# --- Fonts: blocky pixel font for the OSD (SCOPE §3). A bundled font in

font-terminus

# --- System services (SCOPE §3) ---
openssh               # dev access
chrony                # correct time — the clock schedule depends on it
alsa-utils            # amixer for volume/mute

# --- Media library: the USB drive is exFAT (SCOPE §11; user chose exFAT) ---
exfatprogs

# --- VM only: mesa drivers for virtio-gpu (not needed on the Wyse 3040) ---
mesa-dri-gallium      # virtio-gpu DRM/GL driver
mesa-gl               # OpenGL runtime for mpv --vo=gpu inside the VM
