# Rabbit Ears — authoritative package list (Alpine, x86_64).
#
# Source of truth for what the box installs. One package per line; a line
# starting with `#` (or blank) is ignored. Read by alpine/setup.sh.
#
# Names verified against Alpine 3.24 (main + community) via `apk add --simulate`.

# --- Playback: mpv + VAAPI hardware decode + DRM output (SCOPE §3, §4.3) ---
mpv
ffmpeg                # ffprobe for the indexer; mpv's decode backend
libva-intel-driver    # Cherry Trail is Gen8 -> the i965 VAAPI driver
libva-utils           # provides `vainfo` (M1 verifies VAAPI)

# --- App runtime: Python + evdev remote (SCOPE §3, §4.4) ---
python3
py3-evdev

# --- Fonts (SCOPE §3, §4.5). The OSD face is the bundled VCR OSD Mono
#     (system/fonts/, installed to /usr/share/fonts/rabbitears/ by setup.sh);
#     font-dejavu stays as the fontconfig fallback for glyphs it lacks (e.g.
#     accented characters in filenames). font-terminus is out: it's a *bitmap*
#     font (12-32px faces only) and renders nothing at the OSD's 72 px. ---
font-dejavu

# --- System services (SCOPE §3) ---
openssh               # dev access
chrony                # correct time — the clock schedule depends on it
alsa-utils            # amixer for volume/mute

# --- Media library: the USB drive is exFAT (SCOPE §11; user chose exFAT) ---
exfatprogs

# --- VM only: mesa drivers for virtio-gpu (not needed on the Wyse 3040) ---
mesa-dri-gallium      # virtio-gpu DRM/GL driver
mesa-gl               # OpenGL runtime for mpv --vo=gpu inside the VM
